rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // meta/setup — just a flag: "has the workshop password been created yet?"
    // Anyone (even signed-out) can read it, since the app checks this before
    // the user has logged in, to decide whether to show "set a password" or
    // "enter the password". It can only be created once (no updates/deletes),
    // and only by a request that is already signed in as the workshop account
    // (createUserWithEmailAndPassword happens right before this write).
    match /meta/setup {
      allow read: if true;
      allow create: if request.auth != null;
      allow update, delete: if false;
    }

    // workspaces/{uid} — the actual app data (fabrics/products/orders).
    // Only the signed-in workshop account (matching uid) can read or write
    // its own document. Since there's one shared account, this simply means
    // "must be logged in with the workshop password".
    match /workspaces/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }

    // everything else: locked by default
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
